analysis.sql 
WITH sales AS (
  SELECT
    employee,
    SUM(amount) AS revenue
  FROM orders
  GROUP BY employee
)

SELECT
  employee,
  revenue,
  RANK() OVER (
    ORDER BY revenue DESC
  ) AS ranking
FROM sales;
